BOT_TOKEN = None
def set_token(t):
    global BOT_TOKEN
    BOT_TOKEN = t

def validate_init_data(init_data):
    if not init_data or not BOT_TOKEN: return None
    try:
        parsed = dict(parse_qsl(init_data, keep_blank_values=True))
        got = parsed.pop("hash", None)
        if not got: return None
        check = "\n".join(f"{k}={v}" for k,v in sorted(parsed.items()))
        secret = hmac.new(b"WebAppData", BOT_TOKEN.encode(), hashlib.sha256).digest()
        calc = hmac.new(secret, check.encode(), hashlib.sha256).hexdigest()
        if calc != got: return None
        user = json.loads(parsed.get("user","{}"))
        return user if user.get("id") else None
    except: return None

async def _user(request):
    ud = validate_init_data(request.headers.get("X-Init-Data",""))
    if not ud: raise web.HTTPUnauthorized(text="bad init")
    return ud

async def _player(uid):
    return await db.fetchone("SELECT * FROM players WHERE user_id = ?", (uid,))

async def _add_inv(uid, iid, qty=1):
    await db.execute(
        "INSERT INTO inventory (user_id,item_id,quantity) VALUES (?,?,?) "
        "ON CONFLICT(user_id,item_id) DO UPDATE SET quantity=quantity+excluded.quantity",
        (uid, iid, qty))
    await _refresh_item_count(uid)

async def _remove_inv(uid, iid, qty=1):
    await db.execute(
        "UPDATE inventory SET quantity=quantity-? WHERE user_id=? AND item_id=?",
        (qty, uid, iid))
    await db.execute("DELETE FROM inventory WHERE quantity<=0")
    await _refresh_item_count(uid)

async def _has_inv(uid, iid, qty=1):
    r = await db.fetchone(
        "SELECT quantity FROM inventory WHERE user_id=? AND item_id=?", (uid, iid))
    return r and r["quantity"] >= qty

async def _refresh_item_count(uid):
    r = await db.fetchone(
        "SELECT COUNT(*) as c FROM inventory WHERE user_id=? AND quantity>0", (uid,))
    if r:
        await db.execute("UPDATE players SET owned_items_count=? WHERE user_id=?",
                         (r["c"], uid))

async def _check_levelup(uid, p):
    gained = 0
    while p["level"] < MAX_LEVEL:
        need = xp_to_next(p["level"])
        if p["xp"] < need: break
        p["xp"] -= need
        p["level"] += 1
        gained += 1
    if p["level"] >= MAX_LEVEL: p["xp"] = 0
    if gained:
        await db.execute("UPDATE players SET level=?, xp=? WHERE user_id=?",
                         (p["level"], p["xp"], uid))
    return gained

async def _check_achievements(uid, p):
    unlocked = _json_ach(p)
    newly = []
    def add(aid):
        if aid not in unlocked:
            unlocked.append(aid); newly.append(aid)
    if p["balance"] >= 1000000: add("millionaire")
    if p["balance"] >= 1000000000: add("billionaire")
    if p.get("casino_plays",0) >= 10: add("gambler")
    if p.get("crimes_won",0) >= 5: add("criminal")
    if p.get("credit_amount",0) > 0: add("in_debt")
    if p.get("owned_items_count",0) >= 5: add("fashion")
    if p["level"] >= 10: add("level10")
    if p.get("trades_done",0) >= 1: add("trader")
    if "h_gold_crown" in [p.get("head_id")]: add("rich_king")
    if newly:
        await _save_ach(uid, unlocked)
    return newly

async def api_register(request):
    ud = await _user(request); uid = ud["id"]
    if await _player(uid):
        return web.json_response({"error":"already registered"}, status=400)
    body = await request.json()
    nick = (body.get("nickname") or "").strip()
    if not (2 <= len(nick) <= 20):
        return web.json_response({"error":"Ник 2-20 символов"}, status=400)
    referrer_id = body.get("referrer_id")
    if referrer_id:
        try: referrer_id = int(referrer_id)
        except: referrer_id = None
        if referrer_id == uid:
            referrer_id = None
        else:
            # Проверяем, существует ли пригласивший
            ref_p = await db.fetchone("SELECT user_id FROM players WHERE user_id=?", (referrer_id,))
            if not ref_p:
                referrer_id = None

    await db.execute(
        """INSERT INTO players (user_id,nickname,balance,level,xp,location_id,referrer_id)
        VALUES (?,?,?,?,?,?,?)""",
        (uid, nick, START_BALANCE, 1, 0, START_LOCATION, referrer_id))
        
    if referrer_id:
        # Начисляем бонус пригласившему
        await db.execute("UPDATE players SET balance=balance+5000 WHERE user_id=?", (referrer_id,))
    # Стартовый шмот в инвентарь (но не надет — голый!)
    for iid in ["h_cig","b_alco","l_family"]:
        await _add_inv(uid, iid, 1)
    p = await _player(uid)
    return web.json_response({"player": _player_json(p)})

async def api_businesses(request):
    ud = await _user(request); uid = ud["id"]
    p = await _player(uid)
    owned = {b["id"]: b for b in _json_biz(p)}
    now = int(time.time())
    result = []
    total_pending = 0
    for bid, b in BUSINESSES.items():
        info = {**b, "id":bid, "owned":bid in owned}
        if bid in owned:
            lvl = owned[bid].get("level", 1)
            info["level"] = lvl
            mult = 1.0
            if lvl == 2: mult = 1.75
            elif lvl >= 3: mult = 3.0
            real_income = int(b["income"] * mult)
            info["income"] = real_income
            info["last_collect"] = owned[bid].get("last_collect", now)
            elapsed = max(0, min(86400, now - info["last_collect"]))
            info["pending"] = int(real_income * elapsed / 3600)
            total_pending += info["pending"]
        else:
            info["level"] = 0
        result.append(info)
    return web.json_response({"businesses": result, "pending": total_pending})

async def api_collect_business(request):
    ud = await _user(request); uid = ud["id"]
    p = await _player(uid)
    now = int(time.time())
    biz = _json_biz(p)
    total = 0
    for b in biz:
        info = BUSINESSES.get(b["id"])
        if not info: continue
        lvl = b.get("level", 1)
        mult = 1.0
        if lvl == 2: mult = 1.75
        elif lvl >= 3: mult = 3.0
        real_income = int(info["income"] * mult)
        last = b.get("last_collect", now)
        elapsed = max(0, min(86400, now - last))
        earned = int(real_income * elapsed / 3600)
        total += earned
        b["last_collect"] = now
    if total > 0:
        await _save_biz(uid, biz)
        await db.execute("UPDATE players SET balance=balance+? WHERE user_id=?", (total, uid))
    p = await _player(uid)
    return web.json_response({"player": _player_json(p), "collected": total})
async def api_casino_play(request):
    ud = await _user(request); uid = ud["id"]
    body = await request.json()
    game = body.get("game")
    bet = int(body.get("bet", 0))
    if game not in CASINO_GAMES:
        return web.json_response({"error": "Игра"}, status=400)
    if bet <= 0:
        return web.json_response({"error": "Ставка"}, status=400)
    p = await _player(uid)
    if p["balance"] < bet:
        return web.json_response({"error": "Не хватает"}, status=400)
    now_cd = int(time.time())
    if now_cd - p.get("last_casino", 0) < 5:
        return web.json_response({"error": f"Кд {5 - (now_cd - p.get('last_casino', 0))}с"}, status=400)

    win_amount = 0
    result_text = ""

    if game == "coin":
        choice = body.get("choice", "Орёл")
        side = random.choice(["Орёл", "Решка"])
        result_text = f"Монетка: {side}"
        if choice == side:
            win_amount = int(bet * 2)
        else:
            result_text += " — проигрыш"

    elif game == "dice":
        choice = body.get("choice", "Больше 3")
        roll = random.randint(1, 6)
        result_text = f"Кости: выпало {roll}"
        if (choice == "Больше 3" and roll > 3) or (choice == "Меньше 4" and roll < 4):
            win_amount = int(bet * 1.8)
        else:
            result_text += " — проигрыш"

    elif game == "roulette":
        choice = body.get("choice", "red")
        number = body.get("number")  # если ставка на число
        n = random.randint(0, 36)
        color = roulette_color(n)
        result_text = f"Выпало {n} ({color})"
        if choice == "red" and color == "red":
            win_amount = bet * 2
        elif choice == "black" and color == "black":
            win_amount = bet * 2
        elif choice == "even" and n != 0 and n % 2 == 0:
            win_amount = bet * 2
        elif choice == "odd" and n % 2 == 1:
            win_amount = bet * 2
        elif choice == "low" and 1 <= n <= 18:
            win_amount = bet * 2
        elif choice == "high" and 19 <= n <= 36:
            win_amount = bet * 2
        elif choice == "dozen1" and 1 <= n <= 12:
            win_amount = bet * 3
        elif choice == "dozen2" and 13 <= n <= 24:
            win_amount = bet * 3
        elif choice == "dozen3" and 25 <= n <= 36:
            win_amount = bet * 3
        elif choice == "number" and number is not None:
            if int(number) == n:
                win_amount = bet * 36
        if win_amount == 0:
            result_text += " — проигрыш"

    elif game == "slots":
        symbols = ["🍒", "🍋", "🍊", "🔔", "⭐", "💎"]
        a, b, c = random.choices(symbols, k=3)
        result_text = f"{a}{b}{c}"
        if a == b == c == "💎":
            win_amount = int(bet * 50)
            result_text += " — ДЖЕКПОТ!"
        elif a == b == c:
            win_amount = int(bet * 5)
            result_text += " — x5!"
        elif a == b or b == c or a == c:
            win_amount = int(bet * 1.5)
            result_text += " — x1.5"
        else:
            result_text += " — проигрыш"

    net = win_amount - bet
    now_upd = int(time.time())
    await db.execute("UPDATE players SET balance=balance+?, casino_plays=casino_plays+1, last_casino=? WHERE user_id=?",
                     (net, now_upd, uid))
    credit_paid = await _forced_credit_collect(uid) if net > 0 else 0
    p = await _player(uid)
    ach = await _check_achievements(uid, p)
    if win_amount >= 100000 and "big_winner" not in _json_ach(p):
        await _save_ach(uid, _json_ach(p) + ["big_winner"])
        ach.append("big_winner")
    return web.json_response({"player": _player_json(p), "win": win_amount, "net": net,
                              "result": result_text, "new_achievements": ach})

    p = await _player(uid)
    return web.json_response({"player": _player_json(p), "reward": reward, "streak": streak,
                              "credit_paid": credit_paid})

    if give_money: await db.execute("UPDATE players SET balance=balance-? WHERE user_id=?", (give_money, from_user))

                     (from_user, uid))

async def _pvp_do_action(bid, uid, action, part):
    b = await db.fetchone("SELECT * FROM pvp_battles WHERE battle_id=? AND status='active'", (bid,))
    if not b: return {"error":"Бой не активен"}
    if uid not in (b["attacker_id"], b["defender_id"]):
        return {"error":"Не участник"}
    if b["turn"] != uid:
        return {"error":"Не твой ход"}
    if part not in PVP_PARTS:
        return {"error":"Неверная часть"}
    import json as _j
    import random as _r
    log = _j.loads(b["log"] or "[]")
    key = f"pvp_turn_{bid}"
    is_attacker = uid == b["attacker_id"]

    if key not in _PENDING_MOVES:
        _PENDING_MOVES[key] = {"first": {"uid":uid, "action":action, "part":part}}
        next_turn = b["defender_id"] if is_attacker else b["attacker_id"]
        await db.execute("UPDATE pvp_battles SET turn=?, turn_started_at=? WHERE battle_id=?",
                         (next_turn, int(time.time()), bid))
        return {"waiting": True}

    first = _PENDING_MOVES.pop(key)["first"]
    second = {"uid":uid, "action":action, "part":part}
    attacker_move = first if first["uid"] == b["attacker_id"] else second
    defender_move = second if second["uid"] == b["defender_id"] else first
    p1 = await _player(b["attacker_id"])
    p2 = await _player(b["defender_id"])

    async def exchange(atk_move, def_move, atk_player):
        atk_uid = atk_move["uid"]
        is_atk = atk_uid == b["attacker_id"]
        b2 = await db.fetchone("SELECT * FROM pvp_battles WHERE battle_id=?", (bid,))
        def_hp_col = "defender_hp" if is_atk else "attacker_hp"
        atk_damage = b2["attacker_damage"] if is_atk else b2["defender_damage"]
        def_hp = b2[def_hp_col]
        if atk_move["action"] == "attack":
            if def_move["action"] == "defend" and def_move["part"] == atk_move["part"]:
                return {"type":"block","attacker":atk_player["nickname"],
                        "part":PVP_PARTS_NAMES[atk_move["part"]]}, def_hp
            dmg = max(1, atk_damage + _r.randint(-3, 3))
            new_hp = max(0, def_hp - dmg)
            await db.execute(f"UPDATE pvp_battles SET {def_hp_col}=? WHERE battle_id=?", (new_hp, bid))
            return {"type":"hit","attacker":atk_player["nickname"],
                    "part":PVP_PARTS_NAMES[atk_move["part"]],
                    "damage":dmg,"target_hp":new_hp}, new_hp
        return {"type":"defend","player":atk_player["nickname"],
                "part":PVP_PARTS_NAMES[atk_move["part"]]}, def_hp

    r1, _ = await exchange(attacker_move, defender_move, p1)
    log.append(r1)
    b2 = await db.fetchone("SELECT * FROM pvp_battles WHERE battle_id=?", (bid,))
    if b2["attacker_hp"] <= 0 or b2["defender_hp"] <= 0:
        winner_id = b2["attacker_id"] if b2["attacker_hp"] > 0 else b2["defender_id"]
        loser_id = b2["defender_id"] if winner_id == b2["attacker_id"] else b2["attacker_id"]
        await db.execute("UPDATE pvp_battles SET status='finished', log=? WHERE battle_id=?",
                         (_j.dumps(log), bid))
        return await _pvp_finish(winner_id, loser_id, bid)

    r2, _ = await exchange(defender_move, attacker_move, p2)
    log.append(r2)
    b2 = await db.fetchone("SELECT * FROM pvp_battles WHERE battle_id=?", (bid,))
    if b2["attacker_hp"] <= 0 or b2["defender_hp"] <= 0:
        winner_id = b2["attacker_id"] if b2["attacker_hp"] > 0 else b2["defender_id"]
        loser_id = b2["defender_id"] if winner_id == b2["attacker_id"] else b2["attacker_id"]
        await db.execute("UPDATE pvp_battles SET status='finished', log=? WHERE battle_id=?",
                         (_j.dumps(log), bid))
        return await _pvp_finish(winner_id, loser_id, bid)

    await db.execute("UPDATE pvp_battles SET log=?, turn=?, turn_started_at=? WHERE battle_id=?",
                     (_j.dumps(log), b["attacker_id"], int(time.time()), bid))
    return {"round_done": True}

async def api_gang_get(request):
    ud = await _user(request); uid = ud["id"]

    if not member:
        gangs = await db.fetchall(
            "SELECT g.*, (SELECT COUNT(*) FROM gang_members WHERE gang_id=g.gang_id) as members_count "
            "FROM gangs g ORDER BY g.treasury DESC LIMIT 20")
        return web.json_response({"in_gang":False,"gangs":gangs})
    gang = await db.fetchone("SELECT * FROM gangs WHERE gang_id=?", (member["gang_id"],))
    members = await db.fetchall(
        "SELECT gm.user_id, gm.role, p.nickname, p.level FROM gang_members gm "
        "JOIN players p ON gm.user_id = p.user_id WHERE gm.gang_id=?", (gang["gang_id"],))
    territories = await db.fetchall("SELECT * FROM gang_territories")
    now = int(time.time())
    terr_status = []
    for t in GANG_TERRITORIES:
        row = next((x for x in territories if x["territory_id"] == t["id"]), None)
        info = dict(t)
        if row:
            info["gang_id"] = row.get("gang_id")
            info["capturing_gang_id"] = row.get("capturing_gang_id")
            info["capture_end"] = row.get("capture_end")
            info["capture_left"] = max(0, (row.get("capture_end") or 0) - now) if row.get("capture_end") else 0
        else:
            info["gang_id"] = None
            info["capturing_gang_id"] = None
            info["capture_end"] = None
            info["capture_left"] = 0
        terr_status.append(info)
    notifications = []
    for t in terr_status:
        if t["capture_left"] == 0 and t["capturing_gang_id"] and t["capture_end"]:
            capturing = t["capturing_gang_id"]
            was_mine = t["gang_id"] == gang["gang_id"]
            if capturing == gang["gang_id"]:
                notifications.append({"type":"captured","name":t["name"],"income":t["income"],"good":True})
            elif was_mine:
                notifications.append({"type":"lost","name":t["name"],"income":t["income"],"good":False})
            await db.execute(
                "UPDATE gang_territories SET gang_id=?, capturing_gang_id=NULL, capture_end=NULL "

async def _forced_credit_collect(uid):
    """Если кредит просрочен — списать всё что есть. Возвращает сколько списано."""
    try:
        p = await _player(uid)
        debt = p.get("credit_amount", 0) or 0
        due = p.get("credit_due", 0) or 0
        if debt <= 0 or due <= 0:
            return 0
        now = int(time.time())
        if now < due:
            return 0
        if p["balance"] <= 0:
            return 0
        pay = min(p["balance"], debt)
        await db.execute(
            "UPDATE players SET balance = balance - ?, credit_amount = credit_amount - ? WHERE user_id = ?",
            (pay, pay, uid))
        p = await _player(uid)
        if p["credit_amount"] <= 0:
            await db.execute(
                "UPDATE players SET credit_amount = 0, credit_due = 0 WHERE user_id = ?", (uid,))
        return pay
    except Exception as e:
        print(f"⚠️ _forced_credit_collect error: {e}")
        return 0

async def api_pvp_action(request):
    """Атака/защита в PvP."""
    ud = await _user(request); uid = ud["id"]
    body = await request.json()
    bid = int(body.get("battle_id", 0))
    action = body.get("action")  # attack / defend
    part = body.get("part")      # head / torso / legs
    b = await db.fetchone("SELECT * FROM pvp_battles WHERE battle_id=?", (bid,))
    if not b or uid not in (b["attacker_id"], b["defender_id"]) or b["status"] != "active":
        return web.json_response({"error": "Бой недоступен"}, status=400)
    now = int(time.time())
    if now - (b["turn_started_at"] or now) > 30:
        # таймаут хода — пропуск
        await db.execute("UPDATE pvp_battles SET turn=?, turn_started_at=? WHERE battle_id=?",
                         (2 if b["turn"] == 1 else 1, now, bid))
        return web.json_response({"ok": True, "timeout": True})
    # Упрощённая логика: атакующий наносит урон, защитник может защитить часть
    import random
    att = await _player(uid if b["turn"] == 1 else (b["defender_id"] if b["attacker_id"] == uid else b["attacker_id"]))
    dfn = await _player(b["defender_id"] if b["turn"] == 1 else b["attacker_id"])
    # Определяем, кто атакует
    is_attacker = (b["turn"] == 1 and uid == b["attacker_id"]) or (b["turn"] == 2 and uid == b["defender_id"])
    if not is_attacker:
        # Это защита
        await db.execute("UPDATE pvp_battles SET turn=?, turn_started_at=? WHERE battle_id=?",
                         (2 if b["turn"] == 1 else 1, now, bid))
        return web.json_response({"ok": True, "defended": part})
    # Атака
    dmg = random.randint(5, 15)  # базовый урон
    if part == "head":
        dmg = int(dmg * 1.5)
    # Применяем урон к HP защитника
    hp_field = "defender_hp" if b["turn"] == 1 else "attacker_hp"
    max_hp_field = "defender_max_hp" if b["turn"] == 1 else "attacker_max_hp"
    new_hp = max(0, b[hp_field] - dmg)
    await db.execute(f"UPDATE pvp_battles SET {hp_field}=?, turn=?, turn_started_at=? WHERE battle_id=?",
                     (new_hp, 2 if b["turn"] == 1 else 1, now, bid))
    # Проверка победы
    if new_hp <= 0:
        winner_id = uid
        loser_id = dfn["user_id"]
        # Забираем 10% денег
        loser = await _player(loser_id)
        steal = int(loser["balance"] * 0.1)
        await db.execute("UPDATE players SET balance=balance-? WHERE user_id=?", (steal, loser_id))
        await db.execute("UPDATE players SET balance=balance+? WHERE user_id=?", (steal, winner_id))
        await db.execute("UPDATE pvp_battles SET status='finished' WHERE battle_id=?", (bid,))
        return web.json_response({"ok": True, "finished": True, "winner": winner_id, "steal": steal})
    return web.json_response({"ok": True, "damage": dmg, "hp_left": new_hp})

async def api_territory_capture_start(request):
    """Начать захват территории (только leader/deputy)."""
    ud = await _user(request); uid = ud["id"]
    gid, role = await _gang_role(uid)
    if not gid or role not in ("leader", "deputy"):
        return web.json_response({"error": "Только глава или зам"}, status=403)
    body = await request.json()
    tid = int(body.get("territory_id", -1))
    territory = next((t for t in GANG_TERRITORIES if t["id"] == tid), None)
    if not territory:
        return web.json_response({"error": "Территория не найдена"}, status=404)
    # Проверка: нет ли уже активного захвата этой территории
    active = await db.fetchone(
        "SELECT * FROM territory_captures WHERE territory_id=? AND status='active'", (tid,))
    if active:
        return web.json_response({"error": "Территория уже захватывается"}, status=400)
    # Проверка КД на атаку (30 минут)
    last = await db.fetchone(
        "SELECT * FROM territory_captures WHERE territory_id=? AND gang_id=? "
        "ORDER BY started_at DESC LIMIT 1", (tid, gid))
    if last and int(time.time()) - last["started_at"] < 1800:
        return web.json_response({"error": "КД 30 минут"}, status=400)
    # Считаем суммарный урон банды
    total_dmg = await _gang_total_damage(gid)
    # Создаём запись о захвате
    cap_id = await db.execute(
        "INSERT INTO territory_captures (territory_id, gang_id, status, started_at, total_damage) "
        "VALUES (?,?, 'active', ?, ?)",
        (tid, gid, int(time.time()), total_dmg))
    return web.json_response({"ok": True, "capture_id": cap_id})

async def _gang_total_damage(gid):
    """Суммарный урон всех участников банды (из статов)."""
    members = await db.fetchall("SELECT user_id FROM gang_members WHERE gang_id=?", (gid,))
    total = 0
    for m in members:
        p = await _player(m["user_id"])
        if p:
            stats = get_combat_stats(p)
            total += stats.get("damage", 5)
    return total

async def api_territory_capture_resolve(request):
    """Разрешить захват (вызывается по таймеру или вручную)."""
    # В реальном приложении — фоновая задача. Здесь — эндпоинт для теста.
    ud = await _user(request)
    body = await request.json()
    cap_id = int(body.get("capture_id", 0))
    cap = await db.fetchone("SELECT * FROM territory_captures WHERE capture_id=?", (cap_id,))
    if not cap or cap["status"] != "active":
        return web.json_response({"error": "Нет активного захвата"}, status=404)
    # Проверяем, не идёт ли параллельный захват другой бандой
    other = await db.fetchone(
        "SELECT * FROM territory_captures WHERE territory_id=? AND status='active' AND capture_id!=?",
        (cap["territory_id"], cap_id))
    if other:
        # Битва: сравниваем total_damage
        dmg_a = cap["total_damage"]
        dmg_b = other["total_damage"]
        total = dmg_a + dmg_b
        chance_a = dmg_a / total if total > 0 else 0.5
        # Бросок
        if random.random() < chance_a:
            winner = cap
            loser = other
        else:
            winner = other
            loser = cap
        # Победитель забирает территорию
        await db.execute("UPDATE territory_captures SET status='lost' WHERE capture_id=?", (loser["capture_id"],))
        await db.execute("UPDATE territory_captures SET status='won' WHERE capture_id=?", (winner["capture_id"],))
        await db.execute("UPDATE gang_territories SET gang_id=? WHERE territory_id=?",
                         (winner["gang_id"], winner["territory_id"]))
        return web.json_response({"ok": True, "winner": winner["gang_id"], "battle": True})
    else:
        # Никто не мешает — территория переходит
        await db.execute("UPDATE territory_captures SET status='won' WHERE capture_id=?", (cap_id,))
        await db.execute("UPDATE gang_territories SET gang_id=? WHERE territory_id=?",
                         (cap["gang_id"], cap["territory_id"]))
        return web.json_response({"ok": True, "winner": cap["gang_id"], "battle": False})

async def api_referral_claim(request):
    ud = await _user(request); uid = ud["id"]
    p = await _player(uid)
    body = await request.json()
    milestone = int(body.get("milestone", 0))
    
    # Считаем рефералов
    row = await db.fetchone("SELECT COUNT(*) as c FROM players WHERE referrer_id=?", (uid,))
    count = row["c"] if row else 0
    
    import json
    try: claims = json.loads(p.get("ref_claims") or "[]")
    except: claims = []
    
    if milestone in claims:
        return web.json_response({"error": "Бонус уже получен"}, status=400)
    
    rewards = {3: 15000, 5: 30000, 10: 100000}
    if milestone not in rewards:
        return web.json_response({"error": "Неверная награда"}, status=400)
        
    if count < milestone:
        return web.json_response({"error": f"Нужно {milestone} рефералов"}, status=400)
        
    # Выдаем награду
    reward = rewards[milestone]
    claims.append(milestone)
    claims_json = json.dumps(claims)
    
    await db.execute("UPDATE players SET balance=balance+?, ref_claims=? WHERE user_id=?", (reward, claims_json, uid))
    
    return web.json_response({"ok": True, "reward": reward})



async def api_upgrade_business(request):
    ud = await _user(request); uid = ud["id"]
    body = await request.json()
    bid = body.get("business_id")
    b = BUSINESSES.get(bid)
    if not b: return web.json_response({"error":"Бизнес не найден"}, status=400)
    p = await _player(uid)
    biz = _json_biz(p)
    
    target = next((x for x in biz if x["id"] == bid), None)
    if not target: return web.json_response({"error":"У тебя нет этого бизнеса"}, status=400)
    
    lvl = target.get("level", 1)
    if lvl >= 3: return web.json_response({"error":"Максимальный уровень"}, status=400)
    
    cost = int(b["price"] * (1.3 if lvl == 1 else 1.7))
    if p["balance"] < cost: return web.json_response({"error": "Не хватает денег", "required": cost, "balance": p["balance"]}, status=400)
    
    target["level"] = lvl + 1
    await _save_biz(uid, biz)
    await db.execute("UPDATE players SET balance=balance-? WHERE user_id=?", (cost, uid))
    
    p = await _player(uid)
    return web.json_response({"player": _player_json(p), "success": True})

def xp_to_next(level):
    return 100 + (level - 1) * 50

def total_xp_for_level(level):
    return sum(xp_to_next(i) for i in range(1, level))

def crypto_price():
    """Цена коина меняется каждый час."""
    import time, hashlib
    hour = int(time.time() / 3600)
    h = hashlib.md5(f"coin_{hour}".encode()).hexdigest()
    return 500 + (int(h[:4], 16) % 1000)

    dmg = 5

async function api(method, body = {}, retry = true) {
  let res;
  try {
    res = await fetch(`/api/${method}`, {
      method: "POST",
      headers: {"Content-Type":"application/json", "X-Init-Data": initData},
      body: JSON.stringify(body),
    });
  } catch(e) { throw new Error("Нет связи"); }
  if (res.status === 511) throw new Error("Туннель требует пароль");
  if (!res.ok) {
    const err = await res.json().catch(()=>({}));
    const errMsg = err.error || `HTTP ${res.status}`;
    if (retry && method === "register" && errMsg.toLowerCase().includes("already registered")) {
      console.log("Найден битый персонаж — авто-reset");
      try {
        await fetch(`/api/reset`, {
          method: "POST",
          headers: {"Content-Type":"application/json", "X-Init-Data": initData},
          body: JSON.stringify({}),
        });
        return await api(method, body, false);
      } catch(e) { console.warn("Reset failed:", e); }
    }
    throw new Error(errMsg);
  }
  return res.json();
}

function fmtMoney(n) {
  if (n >= 1e9) return (n/1e9).toFixed(2)+"B";
  if (n >= 1e6) return (n/1e6).toFixed(2)+"M";
  if (n >= 1e3) return (n/1e3).toFixed(1)+"K";
  return n.toLocaleString("ru-RU");
}

async function boot() {
  if (!initData) {
    document.body.innerHTML = '<div style="padding:40px;text-align:center;color:#fff;">Открой через Telegram-бота</div>';
    return;
  }
  try {
    const d = await api("me");
    if (!d.registered) { show("register"); renderRegister(); }
    else {
      state.player = d.player;
      state.classInfo = d.class || { name: d.player.class_name, emoji: d.player.class_emoji };
      show("main"); renderPlayer();
      showTab("home");
      if (d.new_achievements?.length) notifyAchievements(d.new_achievements);
      if (d.credit_paid) setTimeout(() => toast(`💸 Банк списал ${fmtMoney(d.credit_paid)}₽`, "error"), 800);

  } catch(e) {
    console.error("boot error:", e);
    toast("Ошибка загрузки: " + e.message, "error");
    show("register"); renderRegister();
  }
}

let timerInterval = null;
function startTimers() {
  if (timerInterval) clearInterval(timerInterval);
  updateTimers();
  timerInterval = setInterval(updateTimers, 1000);
}

function updateTimers() {
  if (state.jobsCache && document.querySelector('.tab-content[data-tab="work"]') && document.querySelector('.tab-content[data-tab="work"]').classList.contains('active')) {
    renderJobs(state.jobsCache);
  }
  const p = state.player;
  if (!p) return;
  // Ежедневный бонус
  const dailyBtn = $("#daily-btn");
  const dailyTimer = $("#daily-timer");
  if (dailyBtn && dailyTimer) {
    const now = Math.floor(Date.now()/1000);
    const cd = p.last_daily ? (p.last_daily + 86400) - now : 0;
    if (cd <= 0) {
      dailyBtn.style.opacity = "1";
      dailyBtn.style.cursor = "pointer";
      dailyTimer.textContent = "✅";
      dailyTimer.classList.remove("hidden");
    } else {
      dailyBtn.style.opacity = "0.6";
      dailyTimer.textContent = formatTime(cd);
      dailyTimer.classList.remove("hidden");
    }
  }
  // Рулетка
  const rouletteTimer = $("#roulette-timer");
  if (rouletteTimer && p.last_roulette !== undefined) {
    const nowR = Math.floor(Date.now()/1000);
    const cdR = p.last_roulette ? (p.last_roulette + 21600) - nowR : 0;
    if (cdR <= 0) {
      rouletteTimer.textContent = "✅";
      rouletteTimer.classList.add("ready");
    } else {
      rouletteTimer.textContent = formatTime(cdR);
      rouletteTimer.classList.remove("ready");
    }
  }
  // Кредит
  renderCreditBanner();
}

function formatTime(sec) {
  sec = Math.max(0, Math.floor(sec));
  const d = Math.floor(sec / 86400);
  const h = Math.floor((sec % 86400) / 3600);
  const m = Math.floor((sec % 3600) / 60);
  const s = sec % 60;
  if (d > 0) return `${d}д ${h}ч`;
  if (h > 0) return `${h}ч ${m}м`;
  if (m > 0) return `${m}м ${s}с`;
  return `${s}с`;
}

    return;

  banner.classList.remove("hidden");
  const now = Math.floor(Date.now()/1000);
  const due = p.credit_due || 0;
  const left = due - now;
  $("#cb-amount").textContent = fmtMoney(debt) + "₽";
  if (left <= 0) {
    banner.classList.add("overdue");
    $("#cb-timer").textContent = "🔴 ПРОСРОЧЕН · списывается автоматически";
  } else {
    banner.classList.remove("overdue");
    $("#cb-timer").textContent = `⏳ Вернуть через ${formatTime(left)}`;
  }
}

function renderAppearance() {
  const container = $("#appearance-list");
  const filterEl = $("#appearance-filter");

  // Mode switch
  let modeHtml = `<div class="mode-switch">
    <button class="mode-btn ${state.appearanceMode==='shop'?'active':''}" onclick="setAppearanceMode('shop')">🛒 Магазин</button>
    <button class="mode-btn ${state.appearanceMode==='inventory'?'active':''}" onclick="setAppearanceMode('inventory')">🎒 Инвентарь</button>
  </div>`;

  // Filters
  let filterHtml = SLOT_FILTERS.map(f=>`<button class="filter-btn ${state.shopFilter===f.id?'active':''}" onclick="setShopFilter('${f.id}')">${f.emoji} ${f.name}</button>`).join("");

  filterEl.innerHTML = modeHtml + filterHtml;

  const p = state.player;
  let listHtml = "";

  if (state.appearanceMode === "shop") {
    const items = state.shopItems.filter(i=>i.slot===state.shopFilter);
    if (items.length === 0) listHtml = '<p style="color:var(--hint);text-align:center;padding:20px">Пусто</p>';
    else listHtml = items.map(it => {
      const canAfford = p.balance >= it.price;
      return `<div class="item-card" style="border-left-color:${it.rarity_color}" onclick="showShopItem('${it.item_id}')">
        <div class="item-preview-mini">${renderCharacterSVG({...emptyOutfit(), [SLOT_KEY[it.slot]]: {item_id: it.item_id}}, {width: 40, height: 80})}</div>
        <div class="item-info">
          <div class="item-name" style="color:${it.rarity_color}">${it.name}</div>
          <div class="item-meta">${it.desc}</div>
        </div>
        <div class="item-price ${canAfford?'':'disabled'}">${fmtMoney(it.price)}₽</div>
      </div>`;
    }).join("");
  } else {
    // Inventory mode
    const equipped = [];
    for (const [slot, item] of Object.entries(p.outfit)) {
      if (item.item_id) equipped.push({slot, item});
    }
    const slotEquipped = equipped.filter(e => SLOT_KEY[e.slot.replace("_id","")] === undefined ? true : SLOT_FILTERS.find(f=>f.id===SLOT_KEY[e.slot] === undefined || true));
    const inv = state.invItems.filter(it => it.slot === state.shopFilter);
    const equippedInSlot = equipped.filter(e => SLOT_KEY[state.shopFilter] === e.slot);

    if (equippedInSlot.length > 0) {
      listHtml += '<div style="font-size:11px;color:var(--hint);margin-bottom:6px;text-transform:uppercase;font-weight:700">Надето</div>';
      listHtml += equippedInSlot.map(e=>`
        <div class="item-card" style="border-left-color:var(--success)" onclick="showEquipped('${e.slot}')">
          <div class="item-preview-mini">${renderCharacterSVG({...emptyOutfit(), [e.slot]: {item_id: e.item.item_id}}, {width: 40, height: 80})}</div>
          <div class="item-info"><div class="item-name">${e.item.name}</div><div class="item-meta">✓ надето</div></div>
          <div style="color:var(--success);font-weight:700">✓</div>
        </div>`).join("");
      listHtml += '<div style="font-size:11px;color:var(--hint);margin:12px 0 6px;text-transform:uppercase;font-weight:700">Сумка</div>';
    }
    if (inv.length === 0) {
      listHtml += '<p style="color:var(--hint);text-align:center;padding:20px">Нет предметов в этом разделе</p>';
    } else {
      listHtml += inv.map(it=>`<div class="item-card" style="border-left-color:${it.rarity_color}" onclick="showInvItem('${it.item_id}')">
        <div class="item-preview-mini">${renderCharacterSVG({...emptyOutfit(), [SLOT_KEY[it.slot]]: {item_id: it.item_id}}, {width: 40, height: 80})}</div>
        <div class="item-info"><div class="item-name" style="color:${it.rarity_color}">${it.name}</div><div class="item-meta">${it.desc}</div></div>
        <div class="item-price">×${it.quantity}</div>
      </div>`).join("");
    }
  }
  container.innerHTML = listHtml;
}

    <div class="job-card ${cur===j.id?'selected':''} ${j.unlocked?'':'locked'}" onclick="selectJob('${j.id}', ${j.unlocked})">

function halveBet() {
  const inp = $("#bet-input");

}

let creditSelectedTerm = null;
function selectCreditTerm(el) {
  document.querySelectorAll(".credit-term").forEach(x => {
    x.style.borderColor = "transparent";
    x.style.background = "var(--bg-3)";
    x.querySelector(".credit-term-check").style.display = "none";
  });
  el.style.borderColor = "var(--success)";

  el.querySelector(".credit-term-check").style.display = "block";
  creditSelectedTerm = {hours: parseInt(el.dataset.hours), pct: parseInt(el.dataset.pct)};
  updateCreditPreview();
}

function updateCreditPreview() {
  const amt = parseInt($("#credit-amt")?.value) || 0;
  const preview = $("#credit-preview");
  const btn = $("#credit-take-btn");
  if (!preview || !btn) return;
  if (!creditSelectedTerm || amt < 1 || amt > 15000) {
    preview.style.display = "none";
    btn.disabled = true;
    return;
  }
  preview.style.display = "block";
  const total = Math.floor(amt * (1 + creditSelectedTerm.pct/100));
  const pct = total - amt;
  $("#cp-term").textContent = creditSelectedTerm.hours + (creditSelectedTerm.hours === 1 ? " час" : creditSelectedTerm.hours < 5 ? " часа" : " часов");
  $("#cp-total").textContent = fmtMoney(total) + "₽";
  $("#cp-pct").textContent = fmtMoney(pct) + "₽";
  btn.disabled = false;
}

let chatInterval = null;
function startChatPolling() {
  if (chatInterval) clearInterval(chatInterval);
  loadChat();
  chatInterval = setInterval(loadChat, 3000);
}
function stopChatPolling() {
  if (chatInterval) { clearInterval(chatInterval); chatInterval = null; }
}
async function loadChat() {
  try {
    const d = await api("chat/get");
    const el = $("#chat-messages");
    if (!el) return;
    const atBottom = el.scrollHeight - el.scrollTop - el.clientHeight < 120;
    if (!d.messages || d.messages.length === 0) {
      el.innerHTML = '<p style="color:var(--hint);text-align:center;padding:40px">Пока нет сообщений. Напиши первым!</p>';
    } else {
      el.innerHTML = d.messages.map(m => {
        let time = "";
        try {
          const iso = m.created_at.replace(" ", "T") + "Z";
          time = new Date(iso).toLocaleTimeString("ru-RU", {hour:"2-digit", minute:"2-digit"});
        } catch(e) {}
        return `<div class="chat-msg">
          <div class="chat-msg-head">
            <span class="chat-nick" onclick="viewPlayer(${m.user_id})">${escapeHtml(m.nickname)}</span>
            <span class="chat-lvl">Ур. ${m.level}</span>
            <span class="chat-time">${time}</span>
          </div>
          <div>${escapeHtml(m.message)}</div>
        </div>`;
      }).join("");
    }
    if (atBottom) el.scrollTop = el.scrollHeight;
  } catch(e) {}
}
function escapeHtml(text) {
  const div = document.createElement("div");
  div.textContent = text == null ? "" : text;
  return div.innerHTML;
}

async function respondPvP(bid, accept) {
  try {
    await api("pvp/respond", {battle_id: bid, accept});
    closeModal();
    if (accept) startPvPUI(bid);
    else toast("Отказался");
  } catch(e) { toast(e.message, "error"); }
}

let pvpCurrentBattle = null;
let pvpTimerInterval = null;
let pvpMyTurnPart = null;
let pvpMyTurnAction = null;

async function startPvPUI(bid) {
  if (pvpTimerInterval) { clearInterval(pvpTimerInterval); pvpTimerInterval = null; }
  pvpCurrentBattle = bid;
  pvpMyTurnPart = null;
  pvpMyTurnAction = null;
  loadPvPState(bid);
  // Авто-обновление каждые 3 сек (для синхронизации)
  pvpTimerInterval = setInterval(() => {
    if (pvpCurrentBattle) loadPvPState(pvpCurrentBattle, true);
    else { clearInterval(pvpTimerInterval); pvpTimerInterval = null; }
  }, 3000);
}

async function loadPvPState(bid, silent = false) {
  try {
    const d = await api("pvp/state", {battle_id: bid});
    if (d.status === "finished") {
      if (pvpTimerInterval) { clearInterval(pvpTimerInterval); pvpTimerInterval = null; }
      pvpCurrentBattle = null;
      closeModal();
      toast("⚔️ Бой окончен!", "success");
      refreshPlayer();
      return;
    }
    if (d.status === "pending") {
      if (!silent) toast("Ждём ответа соперника...");
      return;
    }
    const isMyTurn = d.turn === state.player.user_id;
    const logHtml = (d.log || []).slice(-5).map(l => {
      if (l.type === "hit") return `💥 ${l.attacker} → ${l.part}: -${l.damage}`;
      if (l.type === "block") return `🛡 ${l.attacker}: блок ${l.part}`;
      if (l.type === "defend") return `🛡 ${l.player}: защита ${l.part}`;
      if (l.type === "timeout") return `⏰ ${l.player}: таймаут (защита ${l.part})`;
      return "";
    }).join("<br>");
    const myPart = d.is_attacker ? "attacker" : "defender";
    const enemyPart = d.is_attacker ? "defender" : "attacker";
    const me = d[myPart]; const enemy = d[enemyPart];

    // Таймер
    let timerHtml = "";
    if (isMyTurn) {
      timerHtml = `<div style="text-align:center;padding:8px;background:rgba(46,204,113,.15);border-radius:10px;margin:12px 0;font-weight:700;color:var(--success)">🎯 Твой ход (${d.seconds_left}с)</div>`;
    } else {
      const color = d.seconds_left < 10 ? "var(--danger)" : "var(--hint)";
      timerHtml = `<div style="text-align:center;padding:8px;background:rgba(0,0,0,.3);border-radius:10px;margin:12px 0;font-weight:700;color:${color}">⏳ Ход ${enemy.nickname} (${d.seconds_left}с)</div>`;
    }

    openModal(`
      <h2 style="text-align:center;margin-bottom:12px">⚔️ Гоп-стоп</h2>
      <div class="pvp-card">
        <div class="pvp-hp-row"><span>${enemy.nickname}</span><span>${enemy.hp}/${enemy.max_hp}</span></div>
        <div class="pvp-hp-bar"><div class="pvp-hp-fill" style="width:${enemy.hp/enemy.max_hp*100}%"></div></div>
        <div class="pvp-hp-row" style="margin-top:12px"><span>${me.nickname} (ты)</span><span>${me.hp}/${me.max_hp}</span></div>
        <div class="pvp-hp-bar"><div class="pvp-hp-fill" style="width:${me.hp/me.max_hp*100}%"></div></div>
      </div>
      ${timerHtml}
      ${logHtml ? `<div class="pvp-log">${logHtml}</div>` : ''}
      ${isMyTurn ? `
        <button class="pvp-part-btn" onclick="pvpChooseAction('attack')">⚔️ Атаковать</button>
        <button class="pvp-part-btn" onclick="pvpChooseAction('defend')">🛡 Защищаться</button>
      ` : `<p style="text-align:center;padding:12px;color:var(--hint);font-size:13px">Если противник не ответит за ${d.timeout}с — сработает авто-защита</p>`}
    `);
  } catch(e) {
    if (!silent) toast(e.message, "error");
  }
}

function pvpChooseAction(action) {
  pvpMyTurnAction = action;
  const label = action === "attack" ? "Куда бить?" : "Что защищать?";
  openModal(`
    <h2>${label}</h2>
    <button class="pvp-part-btn" style="margin-top:16px" onclick="pvpDoAction('head')">🎯 Голова</button>
    <button class="pvp-part-btn" onclick="pvpDoAction('torso')">🎯 Торс</button>
    <button class="pvp-part-btn" onclick="pvpDoAction('legs')">🎯 Ноги</button>
    <button class="btn btn-secondary" style="margin-top:12px" onclick="loadPvPState(${pvpCurrentBattle})">⬅️ Назад</button>
  `);
}

async function pvpDoAction(part) {
  try {
    const d = await api("pvp/action", {battle_id: pvpCurrentBattle, action: pvpMyTurnAction, part});
    if (d.finished) {
      openModal(`
        <div style="text-align:center;padding:20px 0">
          <div style="font-size:80px">${d.winner_id === state.player.user_id ? '🏆' : '💀'}</div>
          <h2 style="margin-top:16px">${d.winner_id === state.player.user_id ? 'ПОБЕДА!' : 'Поражение'}</h2>
          <p style="margin-top:12px">${d.winner_id === state.player.user_id ? `Ты отжал ${fmtMoney(d.steal)}₽ у ${d.loser_name}` : `${d.winner_name} отжал у тебя ${fmtMoney(d.steal)}₽`}</p>
          <button class="btn btn-primary" onclick="closeModal();refreshPlayer()" style="margin-top:16px">OK</button>
        </div>`);
      refreshPlayer();
      return;
    }
    if (d.waiting) {
      openModal(`<p style="text-align:center;padding:40px;color:var(--hint)">⏳ Ждём хода противника...</p>`);
      setTimeout(() => loadPvPState(pvpCurrentBattle), 2000);
      return;
    }
    loadPvPState(pvpCurrentBattle);
  } catch(e) { toast(e.message, "error"); }
}

const SLOT_KEY_MAP = {head:"head_id", body:"body_id", legs:"legs_id", right_hand:"right_hand_id", left_hand:"left_hand_id"};
let tradeGiveItems = new Set();
let tradeWantItems = new Set();

async function showTradeOffer(uid, name) {
  try {
    const [myInv, theirInv] = await Promise.all([api("inventory"), api("player_items", {user_id: uid})]);
    tradeGiveItems = new Set();
    tradeWantItems = new Set();
    const myItemsHtml = myInv.items.map(i => `
      <div class="trade-item" data-iid="${i.item_id}" onclick="toggleTradeItem(this, 'give')">
        <div class="trade-item-emoji">${i.emoji}</div>
        <div class="trade-item-name">${i.name}</div>
      </div>`).join("");
    const theirItemsHtml = theirInv.items.length > 0 ? theirInv.items.map(i => `
      <div class="trade-item" data-iid="${i.item_id}" onclick="toggleTradeItem(this, 'want')">
        <div class="trade-item-emoji">${i.emoji}</div>
        <div class="trade-item-name">${i.name}</div>
      </div>`).join("") : '<p style="color:var(--hint);padding:20px;text-align:center">У них пусто</p>';
    openModal(`
      <h2 style="text-align:center;margin-bottom:16px">🤝 Трейд с ${name}</h2>
      <p style="font-weight:700;margin-bottom:6px">📤 Я даю</p>
      <input id="give-money" type="number" placeholder="Деньги ₽" style="width:100%;padding:10px;background:var(--bg-3);border:none;border-radius:8px;color:#fff;margin-bottom:6px" value="0">
      <div class="trade-inventory" id="give-inv">${myItemsHtml || '<p style="color:var(--hint);text-align:center;padding:10px">Твой инвентарь пуст</p>'}</div>
      <p style="font-weight:700;margin:12px 0 6px">📥 Я хочу</p>
      <input id="want-money" type="number" placeholder="Деньги ₽" style="width:100%;padding:10px;background:var(--bg-3);border:none;border-radius:8px;color:#fff;margin-bottom:6px" value="0">
      <div class="trade-inventory" id="want-inv">${theirItemsHtml}</div>
      <div class="trade-summary" id="trade-summary">Выбери предметы</div>
      <button class="btn btn-primary" style="margin-top:12px" onclick="sendTradeOffer(${uid})">🤝 Отправить предложение</button>
    `);
  } catch(e) { toast(e.message, "error"); }
}

function toggleTradeItem(el, kind) {
  const iid = el.dataset.iid;
  el.classList.toggle("selected");
  if (kind === "give") {
    if (el.classList.contains("selected")) tradeGiveItems.add(iid);
    else tradeGiveItems.delete(iid);
  } else {
    if (el.classList.contains("selected")) tradeWantItems.add(iid);
    else tradeWantItems.delete(iid);
  }
  const giveMoney = parseInt($("#give-money")?.value) || 0;
  const wantMoney = parseInt($("#want-money")?.value) || 0;
  $("#trade-summary").innerHTML = `
    <b>Ты даёшь:</b> ${giveMoney > 0 ? fmtMoney(giveMoney)+"₽ " : ""}${tradeGiveItems.size > 0 ? tradeGiveItems.size+" предмет(ов)" : "ничего"}<br>
    <b>Ты хочешь:</b> ${wantMoney > 0 ? fmtMoney(wantMoney)+"₽ " : ""}${tradeWantItems.size > 0 ? tradeWantItems.size+" предмет(ов)" : "ничего"}
  `;
}

// Переопределить sendTradeOffer
async function sendTradeOffer(uid) {
  const give_money = parseInt($("#give-money").value) || 0;
  const want_money = parseInt($("#want-money").value) || 0;
  if (!give_money && !want_money && !tradeGiveItems.size && !tradeWantItems.size) {
    toast("Пустой трейд", "error"); return;
  }
  try {
    await api("trade/send", {
      to_user: uid, give_money,
      give_items: [...tradeGiveItems],
      want_money,
      want_items: [...tradeWantItems]
    });
    closeModal();
    toast("✅ Предложение отправлено", "success");
  } catch(e) { toast(e.message, "error"); }
}

async function loadCases() {
  try {
    const d = await api("cases/list");
    const el = $("#cases-list");
    el.innerHTML = d.cases.map(c => `
      <div class="case-preview-card" onclick="showCaseDetail('${c.case_id}')">
        <div class="case-preview-emoji">${c.emoji}</div>
        <div class="case-preview-info">
          <div class="case-preview-name">${c.name}</div>
          <div class="case-preview-sub">5 уникальных предметов</div>
        </div>
        <div class="case-preview-price">${fmtMoney(c.price)}₽</div>
      </div>
    `).join("");
    state.casesData = d;
  } catch(e) { toast(e.message, "error"); }
}

function showCaseDetail(cid) {
  const c = state.casesData.cases.find(x => x.case_id === cid);
  if (!c) return;
  const bal = state.casesData.balance;
  // Использовать пустой outfit для превью
  const itemsHtml = c.items.map(it => {
    const preview = {...emptyOutfit()};
    const slotKey = SLOT_KEY[it.slot];
    preview[slotKey] = {item_id: it.item_id};
    const stats = [];
    if (it.hp) stats.push(`❤️ +${it.hp} HP`);
    if (it.damage) stats.push(`⚔️ +${it.damage} DMG`);
    if (!stats.length) stats.push("—");
    return `<div class="case-modal-item" style="border-color:${it.rarity_color}">
      <div class="case-modal-item-svg">${renderCharacterSVG(preview, {width: 80, height: 160})}</div>
      <div class="case-modal-item-name" style="color:${it.rarity_color}">${it.name}</div>
      <div class="case-modal-item-chance">${it.chance}%</div>
      <div class="case-modal-item-stats">${stats.join("<br>")}</div>
    </div>`;
  }).join("");
  openModal(`
    <div style="text-align:center;margin-bottom:16px">
      <div style="font-size:56px">${c.emoji}</div>
      <h2>${c.name}</h2>
      <div style="font-size:20px;font-weight:900;color:var(--money);margin-top:4px">${fmtMoney(c.price)}₽</div>
    </div>
    <div class="case-modal-items">${itemsHtml}</div>
    <button class="btn btn-primary" onclick="openCase('${cid}')" ${bal < c.price ? 'disabled' : ''}>
      ${bal < c.price ? '❌ Не хватает денег' : `🎁 Открыть за ${fmtMoney(c.price)}₽`}
    </button>
  `);
}

        <button class="btn btn-primary" onclick="respondPvP(${c.battle_id}, true)">⚔️ Принять бой</button>
        <button class="btn btn-secondary" onclick="respondPvP(${c.battle_id}, false)">❌ Отказаться</button>
      `);
    }
  } catch(e) {}
}

async function respondPvP(bid, accept) {
  try {
    await api("pvp/respond", {battle_id: bid, accept});
    closeModal();
    if (accept) startPvPUI(bid);
    else toast("Отказался");
  } catch(e) { toast(e.message, "error"); }
}

async function startPvPUI(bid) {
  if (pvpTimerInterval) { clearInterval(pvpTimerInterval); pvpTimerInterval = null; }
  pvpCurrentBattle = bid;
  pvpMyTurnPart = null;
  pvpMyTurnAction = null;
  loadPvPState(bid);
  // Авто-обновление каждые 3 сек (для синхронизации)
  pvpTimerInterval = setInterval(() => {
    if (pvpCurrentBattle) loadPvPState(pvpCurrentBattle, true);
    else { clearInterval(pvpTimerInterval); pvpTimerInterval = null; }
  }, 3000);
}

async function loadPvPState(bid) {
  try {
    const d = await api("pvp/state", {battle_id: bid});
    if (d.status === "finished") { closeModal(); toast("Бой окончен!", "success"); return; }
    if (d.status === "pending") { closeModal(); toast("Ждём ответа соперника..."); return; }
    const isMyTurn = d.turn === state.player.user_id;
    const logHtml = (d.log || []).slice(-5).map(l => {
      if (l.type === "hit") return `💥 ${l.attacker} попал в ${l.part} на ${l.damage}`;
      if (l.type === "block") return `🛡 ${l.attacker} блокировал ${l.part}`;
      if (l.type === "defend") return `🛡 ${l.player} защищал ${l.part}`;
      return "";
    }).join("<br>");
    const myPart = d.is_attacker ? "attacker" : "defender";
    const enemyPart = d.is_attacker ? "defender" : "attacker";
    const me = d[myPart]; const enemy = d[enemyPart];
    openModal(`
      <h2 style="text-align:center;margin-bottom:12px">⚔️ Гоп-стоп</h2>
      <div class="pvp-card">
        <div class="pvp-hp-row"><span>${enemy.nickname}</span><span>${enemy.hp}/${enemy.max_hp}</span></div>
        <div class="pvp-hp-bar"><div class="pvp-hp-fill" style="width:${enemy.hp/enemy.max_hp*100}%"></div></div>
        <div class="pvp-hp-row" style="margin-top:12px"><span>${me.nickname} (ты)</span><span>${me.hp}/${me.max_hp}</span></div>
        <div class="pvp-hp-bar"><div class="pvp-hp-fill" style="width:${me.hp/me.max_hp*100}%"></div></div>
      </div>
      ${logHtml ? `<div class="pvp-log">${logHtml}</div>` : ''}
      ${isMyTurn ? `
        <p style="font-weight:700;margin:16px 0 8px">Твой ход — выбери действие:</p>
        <button class="pvp-part-btn" onclick="pvpChooseAction('attack')">⚔️ Атаковать</button>
        <button class="pvp-part-btn" onclick="pvpChooseAction('defend')">🛡 Защищаться</button>
      ` : `<p style="text-align:center;padding:20px;color:var(--hint)">⏳ Ход противника...</p>`}
    `);
    if (!isMyTurn) setTimeout(() => loadPvPState(bid), 2000);
  } catch(e) { toast(e.message, "error"); }
}

function pvpChooseAction(action) {
  pvpMyTurnAction = action;
  const label = action === "attack" ? "Куда бить?" : "Что защищать?";
  openModal(`
    <h2>${label}</h2>
    <button class="pvp-part-btn" style="margin-top:16px" onclick="pvpDoAction('head')">🎯 Голова</button>
    <button class="pvp-part-btn" onclick="pvpDoAction('torso')">🎯 Торс</button>
    <button class="pvp-part-btn" onclick="pvpDoAction('legs')">🎯 Ноги</button>
    <button class="btn btn-secondary" style="margin-top:12px" onclick="loadPvPState(${pvpCurrentBattle})">⬅️ Назад</button>
  `);
}

async function pvpDoAction(part) {
  try {
    const d = await api("pvp/action", {battle_id: pvpCurrentBattle, action: pvpMyTurnAction, part});
    if (d.finished) {
      openModal(`
        <div style="text-align:center;padding:20px 0">
          <div style="font-size:80px">${d.winner_id === state.player.user_id ? '🏆' : '💀'}</div>
          <h2 style="margin-top:16px">${d.winner_id === state.player.user_id ? 'ПОБЕДА!' : 'Поражение'}</h2>
          <p style="margin-top:12px">${d.winner_id === state.player.user_id ? `Ты отжал ${fmtMoney(d.steal)}₽ у ${d.loser_name}` : `${d.winner_name} отжал у тебя ${fmtMoney(d.steal)}₽`}</p>
          <button class="btn btn-primary" onclick="closeModal();refreshPlayer()" style="margin-top:16px">OK</button>
        </div>`);
      refreshPlayer();
      return;
    }
    if (d.waiting) {
      openModal(`<p style="text-align:center;padding:40px;color:var(--hint)">⏳ Ждём хода противника...</p>`);
      setTimeout(() => loadPvPState(pvpCurrentBattle), 2000);
      return;
    }
    loadPvPState(pvpCurrentBattle);
  } catch(e) { toast(e.message, "error"); }
}

async function showTradeOffer(uid, name) {
  try {
    const [myInv, theirInv] = await Promise.all([api("inventory"), api("player_items", {user_id: uid})]);
    tradeGiveItems = new Set();
    tradeWantItems = new Set();
    const myItemsHtml = myInv.items.map(i => `
      <div class="trade-item" data-iid="${i.item_id}" onclick="toggleTradeItem(this, 'give')">
        <div class="trade-item-emoji">${i.emoji}</div>
        <div class="trade-item-name">${i.name}</div>
      </div>`).join("");
    const theirItemsHtml = theirInv.items.length > 0 ? theirInv.items.map(i => `
      <div class="trade-item" data-iid="${i.item_id}" onclick="toggleTradeItem(this, 'want')">
        <div class="trade-item-emoji">${i.emoji}</div>
        <div class="trade-item-name">${i.name}</div>
      </div>`).join("") : '<p style="color:var(--hint);padding:20px;text-align:center">У них пусто</p>';
    openModal(`
      <h2 style="text-align:center;margin-bottom:16px">🤝 Трейд с ${name}</h2>
      <p style="font-weight:700;margin-bottom:6px">📤 Я даю</p>
      <input id="give-money" type="number" placeholder="Деньги ₽" style="width:100%;padding:10px;background:var(--bg-3);border:none;border-radius:8px;color:#fff;margin-bottom:6px" value="0">
      <div class="trade-inventory" id="give-inv">${myItemsHtml || '<p style="color:var(--hint);text-align:center;padding:10px">Твой инвентарь пуст</p>'}</div>
      <p style="font-weight:700;margin:12px 0 6px">📥 Я хочу</p>
      <input id="want-money" type="number" placeholder="Деньги ₽" style="width:100%;padding:10px;background:var(--bg-3);border:none;border-radius:8px;color:#fff;margin-bottom:6px" value="0">
      <div class="trade-inventory" id="want-inv">${theirItemsHtml}</div>
      <div class="trade-summary" id="trade-summary">Выбери предметы</div>
      <button class="btn btn-primary" style="margin-top:12px" onclick="sendTradeOffer(${uid})">🤝 Отправить предложение</button>
    `);
  } catch(e) { toast(e.message, "error"); }
}

function toggleTradeItem(el, kind) {
  const iid = el.dataset.iid;
  el.classList.toggle("selected");
  if (kind === "give") {
    if (el.classList.contains("selected")) tradeGiveItems.add(iid);
    else tradeGiveItems.delete(iid);
  } else {
    if (el.classList.contains("selected")) tradeWantItems.add(iid);
    else tradeWantItems.delete(iid);
  }
  const giveMoney = parseInt($("#give-money")?.value) || 0;
  const wantMoney = parseInt($("#want-money")?.value) || 0;
  $("#trade-summary").innerHTML = `
    <b>Ты даёшь:</b> ${giveMoney > 0 ? fmtMoney(giveMoney)+"₽ " : ""}${tradeGiveItems.size > 0 ? tradeGiveItems.size+" предмет(ов)" : "ничего"}<br>
    <b>Ты хочешь:</b> ${wantMoney > 0 ? fmtMoney(wantMoney)+"₽ " : ""}${tradeWantItems.size > 0 ? tradeWantItems.size+" предмет(ов)" : "ничего"}
  `;
}

// Переопределить sendTradeOffer
async function sendTradeOffer(uid) {
  const give_money = parseInt($("#give-money").value) || 0;
  const want_money = parseInt($("#want-money").value) || 0;
  if (!give_money && !want_money && !tradeGiveItems.size && !tradeWantItems.size) {
    toast("Пустой трейд", "error"); return;
  }
  try {
    await api("trade/send", {
      to_user: uid, give_money,
      give_items: [...tradeGiveItems],
      want_money,
      want_items: [...tradeWantItems]
    });
    closeModal();
    toast("✅ Предложение отправлено", "success");
  } catch(e) { toast(e.message, "error"); }
}

async function renderReferral() {
  const countEl = document.getElementById("ref-count");
  const earnedEl = document.getElementById("ref-earned");
  const linkEl = document.getElementById("ref-link-input");
  const milestonesContainer = document.getElementById("ref-milestones");
  
  if (!countEl) return;

  try {
    const d = await api("referral/info", {});
    
    if (!d || d.error) throw new Error(d?.error || "Пустой ответ сервера");

    countEl.textContent = d.count;
    earnedEl.textContent = fmtMoney(d.earned);
    linkEl.value = d.link;

    const milestones = [
      { count: 3, reward: 15000, icon: "🎁" },
      { count: 5, reward: 30000, icon: "💰" },
      { count: 10, reward: 100000, icon: "👑" }
    ];

    milestonesContainer.innerHTML = "";

    milestones.forEach(m => {
      const isClaimed = d.claims.includes(m.count);
      const isReady = d.count >= m.count;
      
      let btnHtml = "";
      if (isClaimed) {
        btnHtml = `<button class="ref-mile-btn" disabled>✅ Получено</button>`;

        btnHtml = `<button class="ref-mile-btn" onclick="claimRefReward(${m.count})">🎁 Забрать ${fmtMoney(m.reward)}₽</button>`;
      } else {
        btnHtml = `<button class="ref-mile-btn" disabled>Нужно еще ${m.count - d.count} чел.</button>`;
      }

      milestonesContainer.innerHTML += `
        <div class="ref-mile-card ${isClaimed ? 'claimed' : ''}">

async function playCasino() {
  if (state.casinoLocked && Date.now() - state.casinoLockTime < 5000) {
    toast(`⏳ Кд ${Math.ceil((state.casinoLockTime + 5000 - Date.now())/1000)}с`, "error");
    return;
  }
  const betInput = document.getElementById("bet-input");
  const bet = parseInt(betInput?.value) || 0;
  if (bet <= 0) { toast("Введи ставку", "error"); return; }
  if (bet > state.player.balance) { toast("Не хватает денег", "error"); return; }

  const game = state.casinoGame;
  const resEl = document.getElementById('casino-result');
  resEl.textContent = "Крутим...";
  resEl.style.color = "#fff";
  const btn = document.getElementById('btn-play-casino');
  if (btn) btn.disabled = true;

  // Анимация
  if (game === 'coin') {
    const coin = document.getElementById('coin-el');
    if (coin) {
      coin.classList.remove('flip-heads', 'flip-tails');
      void coin.offsetWidth;
      coin.classList.add(Math.random() > 0.5 ? 'flip-heads' : 'flip-tails');
    }
  } else if (game === 'dice') {
    const d = document.getElementById('dice-el');
    if (d) d.classList.add('rolling');
  } else if (game === 'slots') {
    document.querySelectorAll('.slot-reel').forEach(r => r.classList.add('spinning'));
    ['reel1','reel2','reel3'].forEach(id => {
      let html = '';
      for (let j = 0; j < 10; j++) html += `<div class="slot-item">${['🍒','🍋','🍊','🔔','⭐','💎'][Math.floor(Math.random()*6)]}</div>`;
      document.getElementById(id).innerHTML = html;
    });
  }

  try {
    const payload = { game, bet };
    if (game === 'roulette') {
      payload.choice = state.rouletteChoice;
      if (state.rouletteChoice === 'number') payload.number = state.rouletteNumber;
    } else if (game === 'coin') {
      payload.choice = state.coinChoice;
    } else if (game === 'dice') {
      payload.choice = state.diceChoice;
    }
    const d = await api("casino_play", payload);

    await new Promise(r => setTimeout(r, 2000));

    if (game === 'dice') {
      const dEl = document.getElementById('dice-el');
      if (dEl) dEl.classList.remove('rolling');
    }
    if (game === 'slots') {
      document.querySelectorAll('.slot-reel').forEach(r => r.classList.remove('spinning'));
    }

    if (d.win > 0) {
      resEl.textContent = `🎉 ВЫИГРЫШ: ${fmtMoney(d.win)}₽`;
      resEl.style.color = "#2ecc71";
      if (navigator.vibrate) navigator.vibrate([100, 50, 100]);
    } else {
      resEl.textContent = `😢 ПРОИГРЫШ`;
      resEl.style.color = "#e74c3c";
    }

    if (d.player) {
      state.player = d.player;
      document.getElementById('money').textContent = fmtMoney(state.player.balance);
    }

    state.casinoLocked = true;
    state.casinoLockTime = Date.now();
    setTimeout(() => { if (btn) btn.disabled = false; }, 5000);
  } catch(e) {
    resEl.textContent = e.message || "Ошибка сервера";
    toast("Ошибка казино: " + e.message, "error");
    if (btn) btn.disabled = false;
  }
}

async function renderGhetto() {
  const container = document.getElementById('ghetto-map-container');
  if (!container) return;
  container.innerHTML = '<div style="color:#aaa;text-align:center;grid-column:1/-1;">Загрузка...</div>';

  try {
    // Ждём ответ максимум 5 секунд
    const d = await Promise.race([
      api("gang/territories", {}),
      new Promise((_, reject) => setTimeout(() => reject(new Error("Таймаут запроса")), 5000))
    ]);

    container.innerHTML = '';
    if (!d.territories || d.territories.length === 0) {
      container.innerHTML = '<div style="color:#aaa;text-align:center;grid-column:1/-1;">Территории не найдены</div>';
      return;
    }

    d.territories.forEach(t => {
      const ownerName = t.owner ? t.owner.name : "Свободна";
      const ownerColor = t.owner ? "#e67e22" : "#2ecc71";
      const captureInfo = t.capture ? `<div style="font-size:10px;color:#f39c12;margin-top:4px;">⚔️ Захват...</div>` : "";
      
      container.innerHTML += `
        <div class="territory-card" style="background:#1e2a38;border-radius:12px;padding:12px;text-align:center;border:2px solid ${ownerColor};cursor:pointer;box-shadow:0 4px 6px rgba(0,0,0,0.3);" onclick="territoryAction(${t.id}, '${t.name}')">
          <div style="font-size:32px;margin-bottom:4px;">${t.emoji}</div>
          <div style="font-size:14px;font-weight:bold;color:#fff;">${t.name}</div>
          <div style="font-size:12px;color:${ownerColor};margin-top:4px;font-weight:bold;">${ownerName}</div>
          <div style="font-size:11px;color:#8494a5;margin-top:4px;">💰 ${t.income}₽/ч</div>
          ${captureInfo}
        </div>`;
    });
  } catch(e) {
    container.innerHTML = `<div style="color:#e74c3c;text-align:center;grid-column:1/-1;">Ошибка: ${e.message}</div>`;
  }
}

async function territoryAction(tid, name) {
  try {
    const info = await api("gang/info", {});
    if (!info.in_gang) {
      toast("Ты не в банде, чтобы захватывать территории", "error");
      return;
    }
    if (info.role !== "leader" && info.role !== "deputy") {
      toast("Только глава или зам могут захватывать", "error");
      return;
    }
    if (!confirm(`Начать захват территории "${name}"? (Длится 1 час)`)) return;
    
    await api("territory/capture/start", { territory_id: tid });
    toast("⚔️ Захват начат!", "success");
    renderGhetto();
  } catch(e) {
    toast(e.message, "error");
  }
}
